  \pregunta{¿Qué diez distritos reúnen más mujeres reportadas en 2024 con rango de edad especificado?}

  \flujo{\texttt{WHERE}\\Año, sexo y edad}{\texttt{GROUP BY}\\Departamento, provincia, distrito}{\texttt{HAVING SUM >= 20}\\\texttt{ORDER BY personas DESC}}{\texttt{LIMIT 10}\\10 distritos}
